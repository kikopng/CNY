MOBILE-FRIENDLY RETRO SOUND & SPEECH SYNTHESIZER ========================================================== */ class RetroAudio { constructor() { this.ctx = null; this.enabled = true; this.unlocked = false; } init() { if (!this.ctx) { const AudioContext = window.AudioContext || window.webkitAudioContext; if (AudioContext) { this.ctx = new AudioContext(); } } if (this.ctx && this.ctx.state === 'suspended') { this.ctx.resume(); } } // Crucial for Mobile: Plays a silent buffer on first tap to unlock audio unlock() { if (this.unlocked) return; this.init(); if (this.ctx) { try { const buffer = this.ctx.createBuffer(1, 1, 22050); const source = this.ctx.createBufferSource(); source.buffer = buffer; source.connect(this.ctx.destination); source.start(0); if (this.ctx.state === 'suspended') { this.ctx.resume(); } this.unlocked = true; } catch (e) { console.warn('Audio unlock pending user interaction', e); } } // Warm up speech synthesis engine on iOS if ('speechSynthesis' in window) { const emptyUtter = new SpeechSynthesisUtterance(''); window.speechSynthesis.speak(emptyUtter); } } pop() { if (!this.enabled) return; this.init(); if (!this.ctx) return; try { const osc = this.ctx.createOscillator(); const g = this.ctx.createGain(); const now = this.ctx.currentTime; osc.type = 'sine'; osc.frequency.setValueAtTime(400, now); osc.frequency.exponentialRampToValueAtTime(800, now + 0.08); g.gain.setValueAtTime(0.3, now); g.gain.exponentialRampToValueAtTime(0.001, now + 0.08); osc.connect(g); g.connect(this.ctx.destination); osc.start(now); osc.stop(now + 0.09); } catch (e) {} } correct() { if (!this.enabled) return; this.init(); if (!this.ctx) return; try { const notes = [523.25, 659.25, 783.99, 1046.50]; notes.forEach((freq, i) => { const osc = this.ctx.createOscillator(); const g = this.ctx.createGain(); const now = this.ctx.currentTime + (i * 0.07); osc.type = 'triangle'; osc.frequency.setValueAtTime(freq, now); g.gain.setValueAtTime(0.2, now); g.gain.exponentialRampToValueAtTime(0.001, now + 0.2); osc.connect(g); g.connect(this.ctx.destination); osc.start(now); osc.stop(now + 0.22); }); } catch (e) {} } wrong() { if (!this.enabled) return; this.init(); if (!this.ctx) return; try { const now = this.ctx.currentTime; const osc = this.ctx.createOscillator(); const g = this.ctx.createGain(); osc.type = 'sawtooth'; osc.frequency.setValueAtTime(240, now); osc.frequency.linearRampToValueAtTime(140, now + 0.25); g.gain.setValueAtTime(0.2, now); g.gain.exponentialRampToValueAtTime(0.001, now + 0.25); osc.connect(g); g.connect(this.ctx.destination); osc.start(now); osc.stop(now + 0.26); } catch (e) {} } fanfare() { if (!this.enabled) return; this.init(); if (!this.ctx) return; try { const chord = [523.25, 659.25, 783.99, 1046.5, 1318.5]; chord.forEach((freq, idx) => { const osc = this.ctx.createOscillator(); const g = this.ctx.createGain(); const now = this.ctx.currentTime + (idx * 0.09); osc.type = 'triangle'; osc.frequency.setValueAtTime(freq, now); g.gain.setValueAtTime(0.25, now); g.gain.exponentialRampToValueAtTime(0.001, now + 0.35); osc.connect(g); g.connect(this.ctx.destination); osc.start(now); osc.stop(now + 0.36); }); } catch (e) {} } } const sound = new RetroAudio(); // Attach one-time touch listeners to unlock audio immediately on first screen tap ['touchstart', 'touchend', 'pointerdown', 'click'].forEach(evtType => { document.addEventListener(evtType, () => sound.unlock(), { once: true, passive: true }); }); function toggleAudio() { sound.unlock(); sound.enabled = !sound.enabled; document.getElementById('sound-btn').innerText = sound.enabled ? '🔊 Sound: ON' : '🔇 Sound: OFF'; if (sound.enabled) sound.pop(); } // Japanese Voice Engine (Configured for Mobile Safari & Android Chrome) let japaneseVoice = null; function initJapaneseVoices() { if (!('speechSynthesis' in window)) return; const voices = window.speechSynthesis.getVoices(); japaneseVoice = voices.find(v => v.lang === 'ja-JP' || v.lang.startsWith('ja')) || null; } if ('speechSynthesis' in window) { initJapaneseVoices(); window.speechSynthesis.onvoiceschanged = initJapaneseVoices; } function speakJapanese(text) { if (!('speechSynthesis' in window) || !sound.enabled) return; try { sound.unlock(); const utter = new SpeechSynthesisUtterance(text); utter.lang = 'ja-JP'; if (japaneseVoice) utter.voice = japaneseVoice; utter.rate = 0.85; // Slightly slower, clear articulation for learners window.speechSynthesis.speak(utter); } catch (e) { console.warn('SpeechSynthesis error:', e); } }
